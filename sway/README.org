Sway analogue of the Hyprland + hyprframe setup. Same keys, apps, gaps, and
host-specific monitors. Sway has no empty frames: ~$mod+h~ / ~$mod+v~ set the
split for the next window, and ~$mod+x~ merges the focused container into a
sibling (hyprframe ~close_frame~).

* Host script
Runtime hostname switch, same machines as Hyprland.

#+begin_src bash :tangle "~/.config/sway/host.sh" :tangle-mode (identity #o777) :shebang "#!/bin/bash"
  hostname="$(hostname)"

  swaymsg -q corner_radius 12 2>/dev/null || true

  case "$hostname" in
      heimdall)
          swaymsg -q output DP-1 mode 3440x1440@175Hz pos 0 0 scale 1 adaptive_sync off
          swaymsg -q output DP-1 render_bit_depth 10 2>/dev/null || true
          swaymsg -q output DP-1 color_profile srgb 2>/dev/null || true
          if ! pgrep -x openrgb >/dev/null 2>&1; then
              openrgb --profile off.orp >/dev/null 2>&1 &
          fi
          ;;
      red)
          swaymsg -q output eDP-1 mode 1920x1080 pos 0 0 scale 1
          swaymsg -q "bindsym --no-repeat Mod4+Shift+w exec qutebrowser"
          ;;
      poseidon)
          swaymsg -q output DP-2 mode 3840x2160 pos 0 0 scale 1
          swaymsg -q output DP-1 mode 3840x2160 pos 3840 0 scale 1
          ;;
      *)
          swaymsg -q output '*' mode preferred
          ;;
  esac
#+end_src

* Helpers

** Close frame
Move every window in the focused container into the nearest sibling, then
let the empty container collapse. No-op on a lone root leaf.

#+begin_src python :tangle "~/.local/bin/sway-close-frame" :tangle-mode (identity #o777) :shebang "#!/usr/bin/env python3"
  import json
  import subprocess
  import sys


  def tree():
      return json.loads(subprocess.check_output(["swaymsg", "-t", "get_tree", "-r"]))


  def run(cmd):
      subprocess.run(["swaymsg", cmd], check=False, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)


  def walk(node):
      yield node
      for child in node.get("nodes", []) + node.get("floating_nodes", []):
          yield from walk(child)


  def is_window(node):
      if node.get("type") == "workspace":
          return False
      if node.get("nodes") or node.get("floating_nodes"):
          return False
      return bool(node.get("pid") or node.get("app_id") or node.get("window"))


  def parents_of(root):
      mapping = {}
      for node in walk(root):
          for child in node.get("nodes", []) + node.get("floating_nodes", []):
              mapping[child["id"]] = node
      return mapping


  def first_window(node):
      if is_window(node):
          return node
      for child in node.get("nodes", []):
          found = first_window(child)
          if found is not None:
              return found
      return node


  def main():
      root = tree()
      parents = parents_of(root)
      focused = next((n for n in walk(root) if n.get("focused")), None)
      if focused is None:
          return

      parent = parents.get(focused["id"])
      if parent is None:
          return

      if parent.get("layout") in ("tabbed", "stacked"):
          frame = parent
      else:
          frame = focused

      split = parents.get(frame["id"])
      if split is None or split.get("type") in ("output", "root"):
          return

      siblings = [c for c in split.get("nodes", []) if c["id"] != frame["id"]]
      if not siblings:
          return

      frames = split.get("nodes", [])
      idx = next(i for i, c in enumerate(frames) if c["id"] == frame["id"])
      if idx + 1 < len(frames):
          target = frames[idx + 1]
      else:
          target = frames[idx - 1]

      mark = "_sway_frame_target"
      dest = first_window(target)
      run("[con_id=%s] mark --add %s" % (dest["id"], mark))

      windows = [w for w in walk(frame) if is_window(w)]
      windows.sort(key=lambda w: bool(w.get("focused")))
      for window in windows:
          run("[con_id=%s] move window to mark %s" % (window["id"], mark))

      run("unmark %s" % mark)
      run("[con_id=%s] focus" % focused["id"])


  if __name__ == "__main__":
      try:
          main()
      except Exception:
          sys.exit(0)
#+end_src
