#+begin_src emacs-lisp
  ;;; init.el --- Emacs initialization  -*- lexical-binding: t; -*-

  ;; ============================================================
  ;; Bootstrap Elpaca
  ;; ============================================================
  ;; Elpaca's built-in Emacs release table does not yet include 31.1
  ;; (2026-08-24), and this build has a nil emacs-build-time.
  (setq elpaca-core-date
        (list (or (and emacs-build-time
                       (string-to-number (format-time-string "%Y%m%d" emacs-build-time)))
                  20260824)))
  (defvar elpaca-installer-version 0.12)
  (defvar elpaca-directory (expand-file-name "elpaca/" user-emacs-directory))
  (defvar elpaca-builds-directory (expand-file-name "builds/" elpaca-directory))
  (defvar elpaca-sources-directory (expand-file-name "sources/" elpaca-directory))
  (defvar elpaca-order '(elpaca :repo "https://github.com/progfolio/elpaca.git"
                                :ref nil :depth 1 :inherit ignore
                                :files (:defaults "elpaca-test.el" (:exclude "extensions"))
                                :build (:not elpaca-activate)))
  (let* ((repo  (expand-file-name "elpaca/" elpaca-sources-directory))
         (build (expand-file-name "elpaca/" elpaca-builds-directory))
         (order (cdr elpaca-order))
         (default-directory repo))
    (add-to-list 'load-path (if (file-exists-p build) build repo))
    (unless (file-exists-p repo)
      (make-directory repo t)
      (when (<= emacs-major-version 28) (require 'subr-x))
      (condition-case-unless-debug err
          (if-let* ((buffer (pop-to-buffer-same-window "*elpaca-bootstrap*"))
                    ((zerop (apply #'call-process `("git" nil ,buffer t "clone"
                                                    ,@(when-let* ((depth (plist-get order :depth)))
                                                        (list (format "--depth=%d" depth) "--no-single-branch"))
                                                    ,(plist-get order :repo) ,repo))))
                    ((zerop (call-process "git" nil buffer t "checkout"
                                          (or (plist-get order :ref) "--"))))
                    (emacs (concat invocation-directory invocation-name))
                    ((zerop (call-process emacs nil buffer nil "-Q" "-L" "." "--batch"
                                          "--eval" "(byte-recompile-directory \".\" 0 'force)")))
                    ((require 'elpaca))
                    ((elpaca-generate-autoloads "elpaca" repo)))
              (progn (message "%s" (buffer-string)) (kill-buffer buffer))
            (error "%s" (with-current-buffer buffer (buffer-string))))
        ((error) (warn "%s" err) (delete-directory repo 'recursive))))
    (unless (require 'elpaca-autoloads nil t)
      (require 'elpaca)
      (elpaca-generate-autoloads "elpaca" repo)
      (let ((load-source-file-function nil)) (load "./elpaca-autoloads"))))
  (add-hook 'after-init-hook #'elpaca-process-queues)
  ;; elpaca-activate let-binds default-directory to the build directory
  ;; of the package it is finishing, and the init queue ends inside that
  ;; binding. Buffers created then, including the dashboard, inherit it,
  ;; so Emacs comes up in ~/.emacs.d/elpaca/builds/<last-package>.
  ;; Normalize on `elpaca-after-init-hook', which fires once the queue
  ;; has fully drained. `emacs-startup-hook' is too early: the queue
  ;; keeps running after it, so buffers created later (dashboard, etc.)
  ;; would still inherit a builds directory.
  ;; command-line-default-directory is the directory Emacs was launched
  ;; from; it can be nil under emacs --daemon, hence the fallback.
  (add-hook 'elpaca-after-init-hook
            (lambda ()
              (let ((dir (or command-line-default-directory "~/"))
                    (roots (mapcar (lambda (root)
                                     (file-name-as-directory (expand-file-name root)))
                                   (list elpaca-builds-directory
                                         elpaca-sources-directory))))
                (dolist (buf (buffer-list))
                  (with-current-buffer buf
                    (when (and (not buffer-file-name)
                               (let ((here (file-name-as-directory
                                            (expand-file-name default-directory))))
                                 ;; Prefix match: buffers can sit in
                                 ;; subdirectories, e.g. builds/ember-theme/.
                                 (cl-some (lambda (root)
                                            (string-prefix-p root here))
                                          roots)))
                      (setq default-directory dir)))))))
  (elpaca `(,@elpaca-order))
  ;; Real startup time. `emacs-init-time' stops at `after-init-hook',
  ;; but elpaca keeps activating packages and running their config
  ;; after that; this reports the full cost. Delete once measured.
  (add-hook 'elpaca-after-init-hook
            (lambda ()
              (message "init + elpaca queue: %.0fms"
                       (* 1000 (float-time (time-subtract (current-time)
                                                          before-init-time))))))
#+end_src

#+begin_src emacs-lisp
  (use-package ligature
    :config
    (ligature-set-ligatures 't '("www"))
    (ligature-set-ligatures 'eww-mode '("ff" "fi" "ffi"))
    (ligature-set-ligatures 'prog-mode
                          '(;; == === ==== => =| =>>=>=|=>==>> ==< =/=//=// =~
                            ;; =:= =!=
                            ("=" (rx (+ (or ">" "<" "|" "/" "~" ":" "!" "="))))
                            ;; ;; ;;;
                            (";" (rx (+ ";")))
                            ;; && &&&
                            ("&" (rx (+ "&")))
                            ;; !! !!! !. !: !!. != !== !~
                            ("!" (rx (+ (or "=" "!" "\." ":" "~"))))
                            ;; ?? ??? ?:  ?=  ?.
                            ("?" (rx (or ":" "=" "\." (+ "?"))))
                            ;; %% %%%
                            ("%" (rx (+ "%")))
                            ;; |> ||> |||> ||||> |] |} || ||| |-> ||-||
                            ;; |->>-||-<<-| |- |== ||=||
                            ;; |==>>==<<==<=>==//==/=!==:===>
                            ("|" (rx (+ (or ">" "<" "|" "/" ":" "!" "}" "\]"
                                            "-" "=" ))))
                            ;; \\ \\\ \/
                            ("\\" (rx (or "/" (+ "\\"))))
                            ;; ++ +++ ++++ +>
                            ("+" (rx (or ">" (+ "+"))))
                            ;; :: ::: :::: :> :< := :// ::=
                            (":" (rx (or ">" "<" "=" "//" ":=" (+ ":"))))
                            ;; // /// //// /\ /* /> /===:===!=//===>>==>==/
                            ("/" (rx (+ (or ">"  "<" "|" "/" "\\" "\*" ":" "!"
                                            "="))))
                            ;; .. ... .... .= .- .? ..= ..<
                            ("\." (rx (or "=" "-" "\?" "\.=" "\.<" (+ "\."))))
                            ;; -- --- ---- -~ -> ->> -| -|->-->>->--<<-|
                            ("-" (rx (+ (or ">" "<" "|" "~" "-"))))
                            ;; *> */ *)  ** *** ****
                            ("*" (rx (or ">" "/" ")" (+ "*"))))
                            ;; www wwww
                            ("w" (rx (+ "w")))
                            ;; <> <!-- <|> <: <~ <~> <~~ <+ <* <$ </  <+> <*>
                            ;; <$> </> <|  <||  <||| <|||| <- <-| <-<<-|-> <->>
                            ;; <<-> <= <=> <<==<<==>=|=>==/==//=!==:=>
                            ;; << <<< <<<<
                            ("<" (rx (+ (or "\+" "\*" "\$" "<" ">" ":" "~"  "!"
                                            "-"  "/" "|" "="))))
                            ;; >: >- >>- >--|-> >>-|-> >= >== >>== >=|=:=>>
                            ;; >> >>> >>>>
                            (">" (rx (+ (or ">" "<" "|" "/" ":" "=" "-"))))
                            ;; #: #= #! #( #? #[ #{ #_ #_( ## ### #####
                            ("#" (rx (or ":" "=" "!" "(" "\?" "\[" "{" "_(" "_"
                                         (+ "#"))))
                            ;; ~~ ~~~ ~=  ~-  ~@ ~> ~~>
                            ("~" (rx (or ">" "=" "-" "@" "~>" (+ "~"))))
                            ;; __ ___ ____ _|_ __|____|_
                            ("_" (rx (+ (or "_" "|"))))
                            ;; Fira code: 0xFF 0x12
                            ("0" (rx (and "x" (+ (in "A-F" "a-f" "0-9")))))
                            ;; Fira code:
                            "Fl"  "Tl"  "fi"  "fj"  "fl"  "ft"
                            ;; The few not covered by the regexps.
                            "{|"  "[|"  "]#"  "(*"  "}#"  "$>"  "^="))
    (global-ligature-mode t))

  (use-package project
    :ensure nil
    :config
    ;; This use-package form runs before no-littering is activated, and
    ;; project.el reads the list as soon as it is touched. Point it at
    ;; the file no-littering will use, or startup sees a stale list.
    (setq project-list-file
          (expand-file-name "var/project-list.el" user-emacs-directory))
    ;; No `project-remember-projects-under' call here: it walked the
    ;; tree on every startup for nothing. Opening files remembers Git

#+begin_src emacs-lisp
  ;; ============================================================
  ;; Tree-sitter — distro grammars only
  ;; ============================================================
  ;; Emacs 31 is linked to libtree-sitter 0.26 (language ABI 15).
  ;; CachyOS/Arch `tree-sitter-*` packages ship ABI-15 .so files and
  ;; load from /usr/lib. Do not call `treesit-install-language-grammar`
  ;; or treesit-auto: those compile GitHub HEAD, which is what last
  ;; produced ABI mismatches. Languages without a distro grammar stay
  ;; on the classic built-in mode where one exists.
  (use-package treesit
    :ensure nil
    :config
    (setq treesit-font-lock-level 3)
    (dolist (pair '((c-mode          . c-ts-mode)
                    (c++-mode        . c++-ts-mode)
                    (c-or-c++-mode   . c-or-c++-ts-mode)
                    (python-mode     . python-ts-mode)
                    (js-mode         . js-ts-mode)
                    (javascript-mode . js-ts-mode)
                    (sh-mode         . bash-ts-mode)
                    (bash-mode       . bash-ts-mode)
                    (lua-mode        . lua-ts-mode)
                    (js-json-mode    . json-ts-mode)
                    (json-mode       . json-ts-mode)
                    (conf-toml-mode  . toml-ts-mode)
                    (toml-mode       . toml-ts-mode)
                    (elixir-mode     . elixir-ts-mode)
                    (rust-mode       . rust-ts-mode)))
      (let ((lang (pcase (cdr pair)
                    ('c-ts-mode      'c)
                    ('c++-ts-mode    'cpp)
                    ('c-or-c++-ts-mode 'cpp)
                    ('python-ts-mode 'python)
                    ('js-ts-mode     'javascript)
                    ('bash-ts-mode   'bash)
                    ('lua-ts-mode    'lua)
                    ('json-ts-mode   'json)
                    ('toml-ts-mode   'toml)
                    ('elixir-ts-mode 'elixir)
                    ('rust-ts-mode   'rust))))
        (when (and lang (treesit-language-available-p lang))
          (add-to-list 'major-mode-remap-alist pair))))
    (add-to-list 'treesit-language-source-alist
  	       '(odin "https://github.com/tree-sitter-grammars/tree-sitter-odin"))
    (add-to-list 'treesit-language-source-alist
  		 '(glsl "https://github.com/tree-sitter-grammars/tree-sitter-glsl"))
    (add-to-list 'treesit-language-source-alist
  		 '(qmljs "https://github.com/yuja/tree-sitter-qmljs")))
