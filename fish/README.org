#+begin_src fish :tangle "~/.config/fish/config.fish"
  # --- Loop-Proof Auto-Bootstrap Fisher ---
  set -l fisher_path $__fish_config_dir/functions/fisher.fish

  if not test -f $fisher_path
      echo "Fisher not found. Installing..."
      mkdir -p $__fish_config_dir/functions
      curl -sL https://raw.githubusercontent.com/jorgebucaran/fisher/main/functions/fisher.fish -o $fisher_path
      source $fisher_path
      fisher install
      fisher update
  end

  # Disable the default "Welcome to fish" message
  set -g fish_greeting ""

  # --- PATH & Environment ---
  fish_add_path ~/.local/bin
  fish_add_path ~/.local/share/bin
  fish_add_path ~/.cargo/bin
  fish_add_path ~/.opencode/bin

  if test -d /usr/lib/jvm/java-21-openjdk
      set -gx JAVA_HOME /usr/lib/jvm/java-21-openjdk
      fish_add_path $JAVA_HOME/bin
  end

  if test -d /opt/android-sdk
      set -gx ANDROID_HOME /opt/android-sdk
      set -gx ANDROID_SDK_ROOT $ANDROID_HOME
      fish_add_path $ANDROID_HOME/platform-tools
  end

  set -gx EDITOR nvim
  set -gx VISUAL emacs
  set -gx LESS -R
  set -gx GPG_TTY (tty)

  # --- Tool Initializations ---
  if command -q starship
  	starship init fish | source
  end

  if command -q zoxide
  	zoxide init --cmd cd fish | source
  end

  if command -q direnv
  	direnv hook fish | source
  end

  if command -q mise
  	mise activate fish | source
  end

  # --- Modern Aliases ---
  if command -q eza
      alias ls='eza --icons --group-directories-first'
      alias ll='eza -lh --icons --git --grid --group-directories-first'
      alias la='eza -a --icons --group-directories-first'
      alias tree='eza --tree --icons'
  end

  if command -q xh
      alias http='xh'
  end

  if command -q bat
      set -gx MANPAGER 'sh -c "col -bx | bat -l man -p"'
      set -gx MANROFFOPT -c
      alias bathelp='bat --plain --language=help'
  end

  # --- Abbreviations (The "Fish Way") ---
  abbr -a gs  'git status'
  abbr -a ga  'git add'
  abbr -a gc  'git commit'
  abbr -a gp  'git push'
  abbr -a ..  'cd ..'
  abbr -a ... 'cd ../..'

  alias xuu="sudo shutdown -h now"
  alias xur="sudo reboot"
  alias pwheimdall="pactl load-module module-tunnel-sink server=tcp:192.168.0.127"
  alias pwposeidon="pactl load-module module-tunnel-sink server=tcp:10.0.0.20"

  # --- FZF ---
  if command -q fd
      set -gx FZF_DEFAULT_COMMAND 'fd --hidden --follow --exclude .git --color=always'
      set -gx FZF_CTRL_T_COMMAND "$FZF_DEFAULT_COMMAND"
      set -gx FZF_ALT_C_COMMAND 'fd --type d --hidden --follow --exclude .git --color=always'
  end
  set -gx FZF_DEFAULT_OPTS "$FZF_DEFAULT_OPTS --ansi"
  set -gx FZF_CTRL_T_OPTS '--preview "bat --color=always --line-range :200 {} 2>/dev/null || eza --tree --color=always --level=2 {} 2>/dev/null || ls -la {}"'
  set -gx FZF_ALT_C_OPTS '--preview "eza --tree --color=always --level=2 {} 2>/dev/null || ls -la {}"'

  if command -q fzf; and fzf --fish >/dev/null 2>&1
      fzf --fish | source
  else if test -f /usr/share/fish/vendor_conf.d/fzf.fish
      source /usr/share/fish/vendor_conf.d/fzf.fish
  end

  # --- Yazi: q cds out, Q quits in place ---
  if command -q yazi
      function y
          set -l tmp (mktemp -t "yazi-cwd.XXXXXX")
          command yazi $argv --cwd-file="$tmp"
          if test -s "$tmp"
              set -l cwd (command cat -- "$tmp")
              if test -n "$cwd" -a "$cwd" != "$PWD" -a -d "$cwd"
                  builtin cd -- "$cwd"
              end
          end
          command rm -f -- "$tmp"
      end
  end
#+end_src
